const OPENROUTER_CHAT_URL = "https://openrouter.ai/api/v1/chat/completions";
const DEFAULT_OPENROUTER_MODEL = "deepseek/deepseek-chat";

function extractJsonReply(raw: string): string {
  const fenced = raw.match(/```(?:json)?\s*([\s\S]*?)```/i);
  const body = (fenced ? fenced[1] : raw).trim();
  const start = body.search(/[{[]/);
  const end = Math.max(body.lastIndexOf("}"), body.lastIndexOf("]"));
  return start !== -1 && end > start ? body.slice(start, end + 1) : body;
}

async function openRouterChat(prompt: string, jsonMode: boolean): Promise<string> {
  if (!runtime.openRouterApiKey) throw new Error("configurePipelineRuntime({ openRouterApiKey }) first");
  const res = await fetch(OPENROUTER_CHAT_URL, {
    method: "POST",
    headers: {
      Authorization: `Bearer ${runtime.openRouterApiKey}`,
      "Content-Type": "application/json",
      "HTTP-Referer": runtime.openRouterSiteUrl ?? "http://localhost:3000",
      "X-Title": runtime.openRouterAppTitle ?? "forge-pipeline-export",
    },
    body: JSON.stringify({
      model: runtime.openRouterModel ?? DEFAULT_OPENROUTER_MODEL,
      messages: [{ role: "user", content: prompt }],
      temperature: 0,
      max_tokens: 8192,
      provider: { allow_fallbacks: false },
      ...(jsonMode ? { response_format: { type: "json_object" } } : {}),
    }),
  });
  if (!res.ok) throw new Error(`OpenRouter ${res.status}: ${await res.text()}`);
  const data = (await res.json()) as { choices?: { message?: { content?: string } }[] };
  const content = data.choices?.[0]?.message?.content;
  if (!content) throw new Error("OpenRouter returned no message content");
  return content;
}

async function llmStructured<T>(prompt: string, schema: unknown): Promise<T> {
  if (!runtime.openRouterApiKey) throw new Error("configurePipelineRuntime({ openRouterApiKey }) first");
  const schemaJson = JSON.stringify(schema ?? {}, null, 2);
  const base = `${prompt}\n\nRespond with ONLY a JSON value that validates against this JSON Schema:\n${schemaJson}`;
  let lastError = "";
  for (let attempt = 0; attempt < 2; attempt++) {
    const ask =
      attempt === 0
        ? base
        : `${base}\n\nYour previous reply failed validation:\n${lastError}\nReturn corrected JSON only.`;
    const raw = await openRouterChat(ask, true);
    const json = extractJsonReply(raw);
    try {
      return JSON.parse(json) as T;
    } catch (e) {
      lastError = (e as Error).message;
      if (attempt === 1) throw new Error(`structured response parse failed after retry: ${lastError}`);
    }
  }
  throw new Error("structured response failed");
}

async function llmBatchStructured<T>(
  items: unknown[],
  promptFor: (item: unknown) => string,
  schema: unknown,
): Promise<T[]> {
  return Promise.all(items.map((item) => llmStructured<T>(promptFor(item), schema)));
}

 * @kind Assert
 * len(Student Submissions) × len(Marking Questions List) = len(Final Output)
 * @cached true
 */
async function step_Grade_Count_Check(input: SectionGrade[], state: PipelineState): Promise<SectionGrade[]> {
  // Validate pipeline state — passthrough on success
  const __assert = (input, state) => {
  function __deepEqual(a, b) {
    const sortKeys = (v) => {

      if (v !== null && typeof v === "object") {

    if (!(label in expectations)) throw new Error(`assert: no expected output for step "${label}"`);
    return expectations[label];
  };

      key !== undefined && v !== null && typeof v === "object" && !Array.isArray(v)
        ? v[key]
        : v;
    if (Array.isArray(x)) return x.length;
    if (x == null) return 0;
    if (typeof x === "string") return x.length;
    return 1;
  };
  const stripKeys = (v, keys) => {
    const ignore = new Set(Array.isArray(keys) ? keys : [keys]);

  const state: PipelineState = {};

  state["Combine Inputs"] = spine;

  // 5. Assemble Grader Work
  spine = await step_Assemble_Grader_Work(spine!, state);
  state["Assemble Grader Work"] = spine;

  // 6. Flatten Sections
  spine = await step_Flatten_Sections(spine!, state);
  state["Flatten Sections"] = spine;

  // 7. Grade Each Section
  spine = await step_Grade_Each_Section(spine!, state);
  state["Grade Each Section"] = spine;

  // 8. Final Output
  spine = await step_Final_Output(spine!, state);
  state["Final Output"] = spine;

  // 9. Grade Count Check
  spine = await step_Grade_Count_Check(spine!, state);
  state["Grade Count Check"] = spine;

  return spine;
